const NOMBRE_HOJA = "Hoja 1"; function doGet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let hoja = ss.getSheetByName(NOMBRE_HOJA); if (!hoja) { hoja = ss.insertSheet(NOMBRE_HOJA); hoja.appendRow(["Nº Poligono", "Nº Parcela", "Hecho", "Fecha de Registro", "Tratamiento", "Fecha de Registro", "Nombre Parcela"]); } const datos = hoja.getDataRange().getValues(); const resultado = []; for (let i = 1; i < datos.length; i++) { let fD = datos[i][3]; fD = (fD instanceof Date) ? Utilities.formatDate(fD, ss.getSpreadsheetTimeZone(), "dd/MM/yyyy") : (fD ? fD.toString() : "-"); let valTratamiento = datos[i][4]; let tratamientoCheckActivo = (valTratamiento === true || valTratamiento === "TRUE" || (valTratamiento && valTratamiento.toString().toLowerCase() === "true")); let fF = datos[i][5]; fF = (fF instanceof Date) ? Utilities.formatDate(fF, ss.getSpreadsheetTimeZone(), "dd/MM/yyyy") : (fF ? fF.toString() : "-"); let txtHecho = datos[i][2] ? datos[i][2].toString() : ""; resultado.push({ poligono: datos[i][0] !== undefined && datos[i][0] !== "" ? datos[i][0].toString() : "-", id: datos[i][1] !== undefined && datos[i][1] !== "" ? datos[i][1].toString() : "-", c1: txtHecho.includes("1"), c2: txtHecho.includes("2"), c3: txtHecho.includes("3"), fechaD: fD, tratamientoCheck: tratamientoCheckActivo, fechaF: fF, nombre: datos[i][6] || "" }); } return ContentService.createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let hoja = ss.getSheetByName(NOMBRE_HOJA); const datosRecibidos = JSON.parse(e.postData.contents); const rangoActual = hoja.getDataRange(); const filasActuales = rangoActual.getValues(); hoja.clearContents(); hoja.appendRow(["Nº Poligono", "Nº Parcela", "Hecho", "Fecha de Registro", "Tratamiento", "Fecha de Registro", "Nombre Parcela"]); const filasNuevas = datosRecibidos.map((p, index) => { let polimix = (filasActuales[index + 1] && filasActuales[index + 1][0] !== "") ? filasActuales[index + 1][0] : "-"; let parcelaMix = (filasActuales[index + 1] && filasActuales[index + 1][1] !== "") ? filasActuales[index + 1][1] : p.id; let resumenHecho = [p.c1?'1':'', p.c2?'2':'', p.c3?'3':''].filter(Boolean).join(",") || "FALSE"; return [ polimix, parcelaMix, resumenHecho, p.fechaD, p.tratamientoCheck ? "TRUE" : "FALSE", p.fechaF, p.nombre ]; }); if (filasNuevas.length > 0) { hoja.getRange(2, 1, filasNuevas.length, 7).setValues(filasNuevas); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
